# Returns the display year as a positive integer (BC or AD)
adjusted_year_BCAD = {
    value = current_year
    subtract = 5000        # +ve = AD, 0 or -ve = BC

    if = {
        limit = { value <= 0 }
        multiply = -1
        add = 1            # no-year-zero: 5000 -> 1 BC, 4999 -> 2 BC, etc.
    }
}

# Returns 1 if AD, 0 if BC — use this to gate era label in GUI/loc
is_ad_year = {
    value = current_year
    subtract = 5000
    # positive = AD, zero or negative = BC
    # Use: script_value > 0 for AD, script_value <= 0 for BC
}

calendar_setting = {
    value = 0  #Defaulted to zero until set in game - Gregorian
    if = {
        limit = { has_game_rule = alexander_calendar }
        add = 1
    }
    else_if = {
        limit = { has_game_rule = macedonian_calendar }
        add = 2
    }
    else_if = {
        limit = { has_game_rule = roman_calendar }
        add = 3
    }
    else_if = {
        limit = { has_game_rule = egyptian_calendar }
        add = 4
    }
}
alexander_era_dead = {
   value = 0
    if = { 
        limit = { character:MEGAS_ALEXANDROS = { is_alive = no } }
        add = 1
    }
}
