on_game_start = {
	on_actions = {
		on_government_change_start
	}
}

on_government_change_start = {
	effect = {
		if = {
			limit = {
				AND = {
				title:k_croatia = {
					is_title_created = yes
				}
					title:k_croatia.holder ?= {
						game_start_date = 476.9.4
					}
				}
			}
			title:k_croatia.holder ?= {
				if = {
					limit = {
						highest_held_title_tier >= tier_kingdom
						has_dlc_feature = roads_to_power
					}
					change_government = classical_government_ep3
				}
				else_if = {
					change_government = classical_government_ep3
				}
				every_vassal_or_below = {
					limit = {
						OR = {
						has_government = feudal_government
						has_government = republic_government
						}
					}
					if = {
						limit = {
							highest_held_title_tier >= tier_county
							has_dlc_feature = roads_to_power
						}
						change_government = classical_government_ep3
					}
					else_if = {
						limit = {
							highest_held_title_tier >= tier_county
						}
						change_government = classical_government_ep3
					}
				}
			}
		}
		if = {
			limit = {
				AND = {
				title:k_france = {
					is_title_created = yes
				}
					title:k_france.holder ?= {
						game_start_date = 476.9.4
					}
				}
			}
			title:k_france.holder ?= {
				if = {
					limit = {
						highest_held_title_tier >= tier_kingdom
						has_dlc_feature = roads_to_power
					}
					change_government = classical_government_ep3
				}
				else_if = {
					change_government = classical_government_ep3
				}
				every_vassal_or_below = {
					limit = {
						OR = {
						has_government = feudal_government
						has_government = republic_government
						}
					}
					if = {
						limit = {
							highest_held_title_tier >= tier_county
							has_dlc_feature = roads_to_power
						}
						change_government = classical_government_ep3
					}
					else_if = {
						limit = {
							highest_held_title_tier >= tier_county
						}
						change_government = classical_government_ep3
					}
				}
			}
		}
		if = {
			limit = {
				has_dlc_feature = khans_of_the_steppe
				title:e_caspian-pontic_steppe = {
					is_title_created = yes
				}
			}
			title:e_caspian-pontic_steppe.holder ?= {
				change_government = nomad_government
			}
		}
		if = {
			limit = {
				has_dlc_feature = khans_of_the_steppe
			}
			title:e_mongolia = {
				every_in_de_jure_hierarchy = {
					limit = {
						is_title_created = yes
					}
					holder = {
						if = {
							limit = {
								has_government = tribal_government
							}
							change_government = nomad_government
						}
					}
				}
			}
			title:e_siberia = {
				every_in_de_jure_hierarchy = {
					limit = {
						is_title_created = yes
					}
					holder = {
						if = {
							limit = {
								has_government = tribal_government
							}
							change_government = nomad_government
						}
					}
				}
			}
			title:e_volga-ural = {
				every_in_de_jure_hierarchy = {
					limit = {
						is_title_created = yes
					}
					holder = {
						if = {
							limit = {
								has_government = tribal_government
							}
							change_government = nomad_government
						}
					}
				}
			}
			title:e_tartaria = {
				every_in_de_jure_hierarchy = {
					limit = {
						is_title_created = yes
					}
					holder = {
						if = {
							limit = {
								has_government = tribal_government
							}
							change_government = nomad_government
						}
					}
				}
			}
			title:k_oghuz_il = {
				every_in_de_jure_hierarchy = {
					limit = {
						is_title_created = yes
					}
					holder = {
						if = {
							limit = {
								has_government = tribal_government
							}
							change_government = nomad_government
						}
					}
				}
			}
		}
	}
}

							limit = { 
								NOT = { any_held_title = { is_noble_family_title = yes } }
