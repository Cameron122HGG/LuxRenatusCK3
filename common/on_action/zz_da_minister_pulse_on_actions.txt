###############################################################
# Different Administrative — quarterly minister side-effect pulse
#
# Each quarter for every landed character (count+), check if the
# scoped character is an administrative top liege with the special
# ministry. For each filled ministry slot, ~2% chance per quarter
# to trigger that ministry's side-effect event (≈ one event every
# ~12 years per minister, ~3 events/year across all four slots).
# scope.root = the minister (councillor)'s top liege; scope:minister
# = the councillor character.
###############################################################

quarterly_playable_pulse = {
	on_actions = {
		da_minister_quarterly_side_effects
	}
}

da_minister_quarterly_side_effects = {
	effect = {
		if = {
			limit = {
				basileus_has_ministry_access = yes
			}
			# Minister of Personnel (diplomacy)
			if = {
				limit = { exists = cp:minister_personnel }
				cp:minister_personnel = { save_scope_as = minister }
				random = {
					chance = 2
					trigger_event = da_minister_pulse.1
				}
			}
			# Minister of Justice (learning)
			if = {
				limit = { exists = cp:minister_justice }
				cp:minister_justice = { save_scope_as = minister }
				random = {
					chance = 2
					trigger_event = da_minister_pulse.2
				}
			}
			# Minister of Works (stewardship)
			if = {
				limit = { exists = cp:minister_works }
				cp:minister_works = { save_scope_as = minister }
				random = {
					chance = 2
					trigger_event = da_minister_pulse.3
				}
			}
			# Minister Grand Marshal (martial)
			if = {
				limit = { exists = cp:minister_grand_marshal }
				cp:minister_grand_marshal = { save_scope_as = minister }
				random = {
					chance = 2
					trigger_event = da_minister_pulse.4
				}
			}
		}
	}
}
