###############################################################
# Different Administrative — Grand Marshal automatic war aid
#
# Every quarter, any admin ruler at war with a filled Grand
# Marshal council slot has a ~25% chance of an eligible military-
# themed admin vassal (duchy+) being conscripted as an ally into
# all of the liege's active wars. Runs for AI rulers too — player
# presence not required. Picked vassal gets a 1-year cooldown.
#
# Cadence: each quarterly pulse fires the effect once and then
# schedules an additional off-cycle invocation 1-2 months later.
# Net result: ~2 attempts per quarter (8/year) instead of 4/year,
# i.e. double the trigger count without changing the 25% chance.
###############################################################

quarterly_playable_pulse = {
	on_actions = {
		da_grand_marshal_war_aid_pulse
	}
}

da_grand_marshal_war_aid_pulse = {
	effect = {
		da_grand_marshal_war_aid_effect = yes
		# Schedule the off-cycle (half-quarter) follow-up attempt.
		trigger_event = {
			on_action = da_grand_marshal_war_aid_offcycle
			months    = { 1 2 }
		}
	}
}

da_grand_marshal_war_aid_offcycle = {
	effect = {
		da_grand_marshal_war_aid_effect = yes
	}
}
