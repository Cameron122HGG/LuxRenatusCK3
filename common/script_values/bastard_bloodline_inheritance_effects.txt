# Scripted effect for bastard bloodline inheritance
# This can be called directly from AGOT bastard events

# Effect: Apply bloodline inheritance to bastard when secret is revealed
apply_bastard_bloodline_inheritance_effect = {
	# Check if character should receive bloodline inheritance
	if = {
		limit = {
			has_character_flag = royal_bastard
			NOT = { has_character_flag = bloodlines_inherited_from_bastard_revelation }
			any_parent = {
				OR = {
					any_character_trait = { has_trait_flag = male_character_bloodline }
					any_character_trait = { has_trait_flag = female_character_bloodline }
					any_character_trait = { has_trait_flag = dynasty_bloodline }
				}
			}
		}
		
		# Set flag to prevent multiple inheritance
		add_character_flag = bloodlines_inherited_from_bastard_revelation
		
		# Special case for Jon Snow (Stark_7) - add jon_snow_nottarg trait
		if = {
			limit = {
				this = character:Stark_7
			}
            add_trait = the_conqueror
            add_trait = the_conciliator
            add_trait = rogue_prince
            add_trait = rhaenyra
            add_trait = mad_king
            add_trait = rhaegar
            add_trait = lyanna_stark
			add_trait = jon_snow_nottarg
			set_global_variable = {
				name = jon_snow_nottarg
				value = character:Stark_7
			}
		}
		
		# Apply bloodline inheritance based on game rule
		if = {
			limit = { has_game_rule = all_descendants }
			real_father = {
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			mother = {
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
		}
		else_if = {
			limit = { has_game_rule = ck2_version }
			# Father's bloodlines (only patrilineal unless matrilineal marriage)
			father = {
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			# Mother's bloodlines
			mother = {
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			# Matrilineal inheritance
			every_parent = {
				limit = { matrilinear_marriage = yes }
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
		}
		else_if = {
			limit = { has_game_rule = ck2_realfather_version }
			# Real father's bloodlines
			real_father = {
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			# Mother's bloodlines
			mother = {
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			# Matrilineal inheritance for real father
			real_father = {
				limit = { matrilinear_marriage = yes }
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
			mother = {
				limit = { matrilinear_marriage = yes }
				every_character_trait = {
					limit = { has_trait_flag = male_character_bloodline }
					ROOT = { add_trait = prev }
				}
				every_character_trait = {
					limit = { has_trait_flag = female_character_bloodline }
					ROOT = { add_trait = prev }
				}
			}
		}

		# Dynasty bloodlines
		if = {
			limit = {
				any_parent = {
					dynasty ?= ROOT.dynasty
					any_character_trait = { has_trait_flag = dynasty_bloodline }
				}
			}
			every_parent = {
				limit = { dynasty ?= ROOT.dynasty }
				every_character_trait = {
					limit = { has_trait_flag = dynasty_bloodline }
					ROOT = { add_trait = prev }
				}
			}
		}

		# Apply claims based on bloodlines (same as legitimate birth)
		if = {
			limit = { has_trait = the_conqueror }
			add_pressed_claim = title:e_the_iron_throne
			add_pressed_claim = title:k_dragonstone
			add_pressed_claim = title:d_dragonstone
			add_pressed_claim = title:c_dragonstone
			trigger_event = {
				id = targaryen.001
				days = 0
			}
		}

		if = {
			limit = { has_trait = lord_of_tides }
			add_pressed_claim = title:d_driftmark
			add_pressed_claim = title:c_driftmark
		}

		if = {
			limit = { has_trait = celtigar }
			add_pressed_claim = title:d_claw_isle
			add_pressed_claim = title:c_claw_isle
		}

		if = {
			limit = { has_trait = bran_the_builder }
			add_pressed_claim = title:k_the_north
			add_pressed_claim = title:d_winterfell
			add_pressed_claim = title:c_winterfell
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_north
			}
		}

		if = {
			limit = { has_trait = durran_godsgrief }
			add_pressed_claim = title:k_the_stormlands
			add_pressed_claim = title:d_shipbreaker_bay
			add_pressed_claim = title:c_storms_end
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_stormlands
			}
		}

		if = {
			limit = { has_trait = lann_the_clever }
			add_pressed_claim = title:k_the_westerlands
			add_pressed_claim = title:d_casterly_rock
			add_pressed_claim = title:c_casterly_rock
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_westerlands
			}
		}

		if = {
			limit = { has_trait = greenhand }
			add_pressed_claim = title:k_the_reach
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_reach
			}
		}

		if = {
			limit = { has_trait = nymeria }
			add_pressed_claim = title:k_dorne
			add_pressed_claim = title:d_sunspear
			add_pressed_claim = title:c_sunspear
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_dorne
			}
		}

		if = {
			limit = { has_trait = grey_king }
			add_pressed_claim = title:k_the_iron_islands
			add_pressed_claim = title:d_pyke
			add_pressed_claim = title:c_pyke
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_iron_islands
			}
		}

		if = {
			limit = { has_trait = edmyn_tully }
			add_pressed_claim = title:k_the_riverlands
			add_pressed_claim = title:d_riverrun
			add_pressed_claim = title:c_riverrun
		}

		if = {
			limit = { has_trait = artos_arryn }
			add_pressed_claim = title:k_the_vale
			add_pressed_claim = title:d_the_giants_lance
			add_pressed_claim = title:c_the_eyrie
			if = {
				limit = {
					title:e_the_iron_throne = { exists = no }
				}
				add_pressed_claim = title:e_the_vale
			}
		}
	}
}
