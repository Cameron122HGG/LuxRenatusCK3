adjusted_year_BCAD = {
    value = 5000
    subtract = current_year

    save_temporary_value_as = current_adjusted_year

    if = {
      limit = { scope:current_adjusted_year >= 0 }

      add = 1
    }
}
alexander_era_dead = {
    value = 0
    if = { 
        limit = { character:MEGAS_ALEXANDROS = { is_alive = no } }
        add = 1
    }
}

calendar_setting = {
    value = 0  #Defaulted to zero until set in game - Gregorian
    if = {
        limit = { has_game_rule = alexander_calendar }
        add = 1
    }
    else_if = {
        limit = { has_game_rule = macedonian_calendar }
        add = 2
    }
    else_if = {
        limit = { has_game_rule = roman_calendar }
        add = 3
    }
    else_if = {
        limit = { has_game_rule = egyptian_calendar }
        add = 4
    }
}
