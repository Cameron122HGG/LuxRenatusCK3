

LR_test_effect = {
	every_ruler = {
		limit = { 
			is_independent_ruler = yes 
			is_landed = yes
			domain_size > domain_limit
			primary_title.tier > tier_duchy
		}
		save_temporary_scope_as = overlord_to_grant
		every_held_title = {
			limit = {
				NOT = { 
					this = scope:overlord_to_grant.capital_county 
				}
				tier = tier_county
			}
			# Grab the county for ease of reference.
			save_temporary_scope_as = current_county
			scope:current_county.de_jure_liege = {
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder = {
					holder = scope:overlord_to_grant
					change = scope:change
					take_baronies = no
				}
			}
			resolve_title_and_vassal_change = scope:change
			add_to_temporary_list = titles_for_granting
		}
		every_held_title = {
			limit = {
				NOT = { 
					this = scope:overlord_to_grant.capital_county.duchy
				}
				title_capital_county.holder = scope:overlord_to_grant
				tier = tier_duchy
			}
			# Create an elevated peasant to claim the county.
			create_character = {
				location = scope:current_county.title_province
				template = random_vassal_generation_template
				save_temporary_scope_as = local_warlord
			}
			
			save_temporary_scope_as = current_duchy
			every_in_list = {
				list = titles_for_granting
				if = {
					limit = { de_jure_liege = scope:current_duchy }
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = change
						add_claim_on_loss = no
					}				
					change_title_holder_include_vassals = {
						holder = scope:local_warlord
						change = scope:change
						take_baronies = yes
					}
					resolve_title_and_vassal_change = scope:change
				}
			}			

			# Assign the seceding title to the new upstart.
			scope:current_duchy = {
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}				
				change_title_holder_include_vassals = {
					holder = scope:local_warlord
					change = scope:change
					take_baronies = yes
				}
				resolve_title_and_vassal_change = scope:change				
			}
		}
	}
	every_ruler = {
		limit = { 
			is_independent_ruler = yes 
			is_landed = yes
			domain_size > domain_limit
			primary_title.tier < tier_kingdom
		}
		save_temporary_scope_as = overlord_to_grant
		every_held_title = {
			limit = {
				NOT = { 
					this = scope:overlord_to_grant.capital_county
					this.duchy = scope:overlord_to_grant.primary_title
				}
				tier = tier_county
			}
			# Grab the county for ease of reference.
			save_temporary_scope_as = current_county
			# Create an elevated peasant to claim the county.
			create_character = {
				location = scope:current_county.title_province
				template = random_vassal_generation_template
				save_temporary_scope_as = local_warlord
			}
			# Assign the seceding title to the new upstart.
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:current_county = {
				change_title_holder = {
					holder = scope:local_warlord
					change = scope:change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}
