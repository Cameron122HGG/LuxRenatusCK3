TFE_split_empire_effect = {
	get_title = title:h_western_roman_empire
	get_title = title:h_eastern_roman_empire
	scope:roman_split_secondary_heir = {
		create_cadet_branch = yes
	}
	title:h_roman_empire = {
		every_in_de_jure_hierarchy = {
			limit = { tier = tier_kingdom }
			if = {
				limit = { title_capital_county = { title_province = { geographical_region = custom_potential_wre } } }
				set_de_jure_liege_title = title:h_western_roman_empire
			}
			else = {
				set_de_jure_liege_title = title:h_eastern_roman_empire
			}
			if = {
				limit = {
					exists = yes
					holder = { is_independent_ruler = no }
					holder.liege = title:h_roman_empire.holder
				}
				holder = { add_character_flag = not_indy }
			}
		}
	}
	scope:roman_split_secondary_heir = {
		get_title = title:c_byzantion
		title:c_kalliopolis = { add_to_list = transfer_ere_list }
		title:c_brysis = { add_to_list = transfer_ere_list }
		title:d_thrace = { add_to_list = transfer_ere_list }
		title:k_thrace = { add_to_list = transfer_ere_list }
		every_in_list = {
			list = transfer_ere_list
			if = {
				limit = {
					is_title_created = yes
					holder = title:h_roman_empire.holder
				}
			}
			save_scope_as = transfer_title
			scope:transfer_title = {
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder_include_vassals = {
					holder = scope:roman_split_secondary_heir
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		title:h_eastern_roman_empire = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder_include_vassals = {
				holder = scope:roman_split_secondary_heir
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
	scope:roman_split_primary_heir = {
		get_title = title:c_lombardia
		get_title = title:c_roma
		if = {
			limit = {
				AND = {
					title:k_italia_annonariae = {
						is_title_created = yes
					}
					title:h_roman_empire.holder = title:k_italia_annonariae.holder
				}
			}
			title:k_italia_annonariae = {
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder_include_vassals = {
					holder = PREV
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		title:h_western_roman_empire = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder_include_vassals = {
				holder = PREV
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
	}

	set_designated_heir = scope:roman_split_primary_heir

	title:h_roman_empire = {
		holder = { 
			remove_trait = augustus
			every_held_title = {
				limit = { 
					OR = {
						tier = tier_empire
						is_noble_family_title = yes
					}
				}
				holder = { destroy_title = PREV }
			}
		}
	}

	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = change
		add_claim_on_loss = no
	}
	root = {
		change_liege = {
			liege = scope:roman_split_primary_heir
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
		
	depose = yes

	title:h_western_roman_empire.holder = {
		every_held_title = {
			limit = {
				trigger_if = {
					limit = {
						exists = title_province
					}
					title_province = {
						geographical_region = custom_potential_ere
					}
				}
				trigger_else = {
					title_capital_county = {
						title_province = {
							geographical_region = custom_potential_ere
						}
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder_include_vassals = {
				holder = title:h_eastern_roman_empire.holder
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
		every_vassal = {
			limit = {
				capital_county = {
					title_province = {
						geographical_region = custom_potential_ere
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = title:h_eastern_roman_empire.holder
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
		set_variable = {
			name = mil_comp
			value = 1500
		}
		set_variable = {
			name = civic_comp
			value = 1500
		}
		set_variable = {
			name = diplo_comp
			value = 1500
		}
		if = {
			limit = {
				has_dlc_feature = roads_to_power
			}
			change_government = administrative_government
			if = {
				limit = { 
					NOT = { any_held_title = { is_noble_family_title = yes } } 
					highest_held_title_tier >= tier_duchy
				}
				if = {
					limit = { is_house_head = no }
					found_cadet_house_decision_effect = { CHARACTER = title:h_western_roman_empire.holder }
				}
				create_noble_family_effect = yes
				domicile ?= { set_up_domicile_estate_effect = yes }
			}
		}
		else = {
			change_government = imperial_government
		}
	}
	title:h_eastern_roman_empire.holder = {
		set_variable = {
			name = mil_comp
			value = 1500
		}
		set_variable = {
			name = civic_comp
			value = 1500
		}
		set_variable = {
			name = diplo_comp
			value = 1500
		}
		if = {
			limit = {
				has_dlc_feature = roads_to_power
			}
			change_government = administrative_government
			if = {
				limit = { 
					NOT = { any_held_title = { is_noble_family_title = yes } } 
					highest_held_title_tier >= tier_duchy
				}
				if = {
					limit = { is_house_head = no }
					found_cadet_house_decision_effect = { CHARACTER = title:h_eastern_roman_empire.holder }
				}
				create_noble_family_effect = yes
				domicile ?= { set_up_domicile_estate_effect = yes }
			}
		}
	}
	scope:split_emperor = {
		every_child = {
			if = {
				limit = {
					has_trait = augustus
				}
				remove_trait = augustus
			}
		}
	}
	title:h_eastern_roman_empire.holder = { remove_trait = augustus }
	title:h_western_roman_empire.holder = { remove_trait = augustus }
	every_ruler = {
		limit = {
			is_independent_ruler = yes
			has_character_flag = not_indy
			primary_title.empire = {
				exists = yes
			}
		}
		save_scope_as = vassal_ruler
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:vassal_ruler = {
			change_liege = {
					liege = primary_title.empire.holder
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	every_ruler = {
		limit = { has_character_flag = not_indy }
		remove_character_flag = not_indy
	}
	if = {
		limit = {
			title:h_western_roman_empire = {
				holder = { government_has_flag = government_is_administrative }
				has_title_law = imperial_elective_succession_law
			}
		}
		title:h_western_roman_empire = {
			clear_title_laws = yes
		}
	}
	if = {
		limit = {
			title:h_eastern_roman_empire = {
				holder = { government_has_flag = government_is_administrative }
				has_title_law = imperial_elective_succession_law
			}
		}
		title:h_eastern_roman_empire = {
			clear_title_laws = yes
		}
	}
}
