###Lux Bloodline Inheritance Effects - for 历史血脉GUI###

add_trait_bloodline_alexander = {
	add_trait = bloodline_alexander
	every_child = {
		even_if_dead = yes
		if = {
			limit = {
				OR = {
					AND = {
						OR = { has_game_rule = ck2 has_game_rule = ck2_realfather }
						is_male = yes
					}
					has_game_rule = all_descendants
					has_game_rule = all_descendants_realfather
				}
			}
			add_trait_bloodline_alexander = yes
		}
	}
}
