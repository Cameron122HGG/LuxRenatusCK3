# Trigger to enable ministry features in script.
# DA edit: special council (ministry slots, council window flavor) only for
# INDEPENDENT administrative rulers. Vassal admins keep the vanilla council.
# Vanilla celestial behavior (China hegemon) is preserved.
tgp_has_access_to_ministry_trigger = {
	OR = {
		# Vanilla path (celestial / China)
		AND = {
			has_title = title:h_china
			government_has_flag = government_is_celestial
		}
		# DA: any independent administrative ruler
		AND = {
			government_has_flag = government_is_roman_autocratic
			is_independent_ruler = yes
		}
	}
}

# Alias kept so the renamed council-position files keep working.
basileus_has_ministry_access = {
	tgp_has_access_to_ministry_trigger = yes
}

tgp_can_be_a_minister_trigger = {
	can_be_councillor_basics_trigger = yes
	# Has to be celestial/admin or a non-ruler. The celestial flag MUST stay here,
	# otherwise Heavenly Empire (celestial/merit) rulers are excluded and the council
	# candidate pool collapses to family members / courtiers only.
	OR = {
		is_ruler = no
		government_has_flag = government_is_celestial
		government_has_flag = government_is_roman_autocratic
	}
	OR = {
		has_variable = can_be_councillor_override
		AND = {
			is_female = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = male_only_law
					has_realm_law = male_preference_law
				}
			}
		}
		AND = {
			is_male = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = female_only_law
					has_realm_law = female_preference_law
				}
			}
		}
	}
}

tgp_can_be_grand_marshal_trigger = {
	can_be_councillor_basics_trigger = yes
	# Has to be celestial/admin or a non-ruler (see note on tgp_can_be_a_minister_trigger).
	OR = {
		is_ruler = no
		government_has_flag = government_is_celestial
		government_has_flag = government_is_roman_autocratic
	}
	can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = $COURT_OWNER$ }
}

# DA edit: All special-council triggers now require the admin character to be
# the TOP LIEGE (independent ruler). Admin vassals fall through to vanilla.

# Legacy alias — true for any independent admin ruler.
dominus_is_basileus_trigger = {
	government_has_flag = government_is_roman_autocratic
	is_independent_ruler = yes
}

# Trigger used for position NAMES dispatch in council_positions files.
# Fires only for: independent admin ruler / their councillor / their landed titles.
dominus_is_byzantine_ruler_trigger = {
	OR = {
		# Case 1: Character is an independent admin ruler
		AND = {
			is_character = yes
			is_ruler = yes
			is_independent_ruler = yes
			government_has_flag = government_is_roman_autocratic
		}
		# Case 2: Landed title whose holder is an independent admin ruler
		AND = {
			is_landed_title = yes
			exists = this.holder
			this.holder = {
				is_independent_ruler = yes
				government_has_flag = government_is_roman_autocratic
			}
		}
		# Case 3: Councillor serving an independent admin liege/court owner
		AND = {
			is_character = yes
			is_councillor = yes
			liege_or_court_owner = {
				is_independent_ruler = yes
				government_has_flag = government_is_roman_autocratic
			}
		}
	}
}

# Legacy alias.
# Previously used to switch tooltips into Byzantine prose. Now disabled so
# every admin top liege uses the generic ministerial (`_tgp_desc`) fallback
# in the customizable_localization dispatchers.
dominus_is_byzantine_ministry_trigger = {
	always = no
}

# Court trigger (For observers viewing an admin top-liege's court).
dominus_is_byzantine_court_trigger = {
	OR = {
		dominus_is_byzantine_ruler_trigger = yes
		AND = {
			NOT = { exists = this }
			any_player = {
				government_has_flag = government_is_roman_autocratic
				is_independent_ruler = yes
			}
		}
	}
}

# DA: extension of vanilla tgp_celestial_military_governor_can_join_war_trigger.
# Admin vassals only qualify if their theme contract is military, frontier or
# naval (the war-fighting themes). Counties and civilian/imperial themes are
# filtered out of the candidate list.
tgp_celestial_military_governor_can_join_war_trigger = {
	save_temporary_scope_as = joining_governor
	OR = {
		vassal_contract_has_flag = celestial_military_appointment
		vassal_contract_has_flag = meritocratic_military_appointment
		has_title = title:e_minister_grand_marshal
		AND = {
			is_landed = yes
			government_has_flag = government_is_roman_autocratic
			highest_held_title_tier >= tier_duchy
			OR = {
				vassal_contract_has_flag = admin_theme_military
				vassal_contract_has_flag = admin_theme_frontier
				vassal_contract_has_flag = admin_theme_naval
			}
		}
	}
	NOR = {
		scope:joining_governor = $TARGET_GOVERNOR$
		scope:joining_governor = $MINISTER_OF_WAR$
		any_war_ally = { this = $TARGET_GOVERNOR$ }
		is_at_war_with = $TARGET_GOVERNOR$
		is_at_war_with = $MINISTER_OF_WAR$
		$TARGET_GOVERNOR$ = {
			any_war_enemy = {
				OR = {
					is_vassal_of = scope:joining_governor
					is_liege_or_above_of = scope:joining_governor
					is_allied_to = scope:joining_governor
					is_tributary_of = scope:joining_governor
					is_at_war_with = scope:joining_governor
				}
			}
		}
	}
}
