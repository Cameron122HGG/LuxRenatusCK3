@imperial_policy_laws_cooldown_value = 10
@imperial_policy_cost_value = 500

gaul_imperial_policy_laws = {
	default = gaul_manor_reform_law
	cumulative = no
	flag = imperial_policy

	gaul_defense_mobilization_law = {
		flag = defense_mobilization_law

		can_keep = {
			realm_law_gaul_use_imperial_policy_trigger = yes
			trigger_if = {
				limit = { is_independent_ruler = no }
				liege = { has_realm_law = gaul_defense_mobilization_law }
			}
		}

		can_pass = {
			trigger_if = {
				limit = {
					top_liege = this
				}
				custom_tooltip = {
					text = imperial_policy_laws_cooldown_desc
					NOT = { has_variable = imperial_policy_laws_cooldown }
				}
			}
			tgp_gaul_defense_mobilization_valid_trigger = yes
		}

		on_pass = {
			## Set cooldown
			set_variable = {
				name = imperial_policy_laws_cooldown
				years = @imperial_policy_laws_cooldown_value
			}
			tgp_imperial_policy_event_effect = yes
		}

		modifier = {
			same_heritage_county_advantage_add = 5
			men_at_arms_maintenance = -0.15
			fort_level = 1
			hostile_raid_time = 0.15
		}
		flag = governors_may_recruit_gallic_horse_archers
		flag = cheaper_gallic_domicile_military_buildings

		pass_cost = {
			prestige = @imperial_policy_cost_value
		}

		ai_will_do = {
			value = 250
		}
	}

		}

		modifier = {
			men_at_arms_recruitment_cost = 0.25
			gaul_feudal_government_vassal_opinion = -15
			development_growth_factor = 0.15
		}

		pass_cost = {
			prestige = @imperial_policy_cost_value
		}

		ai_will_do = {
			value = 200
			if = {
				limit = {
					ai_has_warlike_personality = yes
				}
				value = -50
			}
			if = {
				limit = { government_has_flag = government_is_gaul_feudal }
				add = -1000
			}
		}

					limit = {
						NOR = {
							has_realm_law = gallic_bureaucracy_2
							has_realm_law = gallic_bureaucracy_3
						}
					}
					has_realm_law = gallic_bureaucracy_2
				}
				trigger_if = {
					limit = {
						NOR = {
							has_realm_law = gallic_bureaucracy_2
							has_realm_law = gallic_bureaucracy_3
						}
					}
					has_realm_law = gallic_bureaucracy_3
				}
				is_at_war = no
			}
		}

		on_pass = {
			## Set cooldown
			set_variable = {
				name = imperial_policy_laws_cooldown
				years = @imperial_policy_laws_cooldown_value
