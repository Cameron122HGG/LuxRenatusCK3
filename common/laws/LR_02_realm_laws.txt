@gallic_bureaucracy_cooldown_years_1 = 10
@gallic_bureaucracy_cooldown_years_2 = 20
@gallic_bureaucracy_cooldown_years_3 = 30

gallic_bureaucracy = { # A variant of Imperial Bureaucracy for Gallic Admin Government
	default = gallic_bureaucracy_1
	cumulative = yes
	flag = realm_law
	
	gallic_bureaucracy_0 = {
		modifier = {
			belligerent_opinion = -10
			barons_and_minor_landholders_opinion = 20
			glory_hound_opinion = 10
			parochial_opinion = 10
			courtly_opinion = 5
		}
		flag = uses_gallic_bureaucracy
		flag = titles_cannot_leave_realm_on_succession # Hardcoded flag
		flag = can_designate_heirs

		can_keep = {
			realm_law_use_gallic_bureaucracy = yes
			trigger_if = {
				limit = { is_independent_ruler = no }
				liege = { has_realm_law = gallic_bureaucracy_0 }
			}
		}
		
		can_pass = {
			trigger_if = {
				limit = {
					is_independent_ruler = no
					liege = { has_realm_law = gallic_bureaucracy_0 }
				}
				custom_tooltip = {
					text = gallic_bureaucracy_pending_law_change
					always = no
				}
			}
		}

		on_pass = {
			# Remove modifiers.
			remove_law_related_modifiers_effect = yes
		}
	}
	
	gallic_bureaucracy_1 = {
		modifier = {
			belligerent_opinion = 5
			barons_and_minor_landholders_opinion = -30
			glory_hound_opinion = -15
			parochial_opinion = -15
			courtly_opinion = -5
			minority_opinion = -10
		}
		flag = uses_celestial_bureaucracy
		flag = title_revocation_allowed
		flag = vassal_retraction_allowed
		flag = can_change_partition_succession_laws
		flag = diarchs_want_to_subsidise_without_this_flag
		flag = can_designate_heirs
		flag = can_have_tributaries

		can_keep = {
			realm_law_use_gallic_bureaucracy = yes
			trigger_if = {
				limit = { is_independent_ruler = no }
				liege = { has_realm_law = gallic_bureaucracy_1 }
			}
		}

		can_pass = {
			trigger_if = {
				limit = {
					has_realm_law = gallic_bureaucracy_0
					is_independent_ruler = yes
				}
				custom_description = {
					subject = root
					text = "has_gallic_bureaucracy_cooldown"
					NAND = {
						has_variable = gallic_bureaucracy_cooldown
						NOT = {
							culture = { has_innovation = innovation_all_things }
						}
					}
				}
			}
			trigger_if = {
				limit = {
					is_independent_ruler = no
					liege = { has_realm_law = gallic_bureaucracy_1 }
				}
				custom_tooltip = {
					text = gallic_bureaucracy_pending_law_change
					always = no
				}
			}
			can_pass_law_ca1_trigger = yes
		}
		
		pass_cost = {
			prestige = {
				if = {
					limit = {
						NOR = {
							has_realm_law = gallic_bureaucracy_2
							has_realm_law = gallic_bureaucracy_3
						}
					}
					add = increase_crown_authority_prestige_cost
				}
			}
		}
		
		on_pass = {
			# Increment cooldown breaker tally, if appropriate.
			calculate_authority_cooldown_break_effect = yes
			# Set regular cooldown.
			set_variable = {
				name = gallic_bureaucracy_cooldown
				years = @gallic_bureaucracy_cooldown_years_1
			}
			# Remove modifiers.
			remove_law_related_modifiers_effect = yes
		}

		ai_will_do = {
			if = {
				limit = { has_realm_law = gallic_bureaucracy_0 }
				value = 1
			}
		}
	}
	
	gallic_bureaucracy_2 = {
		modifier = {
			belligerent_opinion = 10
			barons_and_minor_landholders_opinion = -10
			glory_hound_opinion = -10
			parochial_opinion = -10
			courtly_opinion = -5
			zealot_opinion = -10
			minority_opinion = -10
			vassal_tax_contribution_mult = 0.1
			vassal_levy_contribution_mult = 0.1
			republic_government_tax_contribution_add = 0.05
		}
		flag = uses_gallic_bureaucracy
		flag = vassal_refusal_is_treason
		flag = can_change_succession_laws
		flag = admin_change_vassal_gov_cheaper
		flag = can_designate_heirs
		flag = vassal_internal_wars_banned
		flag = can_have_tributaries
		
		can_keep = {
			realm_law_use_gallic_bureaucracy = yes
			trigger_if = {
				limit = {
					this = title:h_china.holder
				}
				situation:dynastic_cycle ?= {
					NOR = {
						situation_current_phase = situation_dynastic_cycle_phase_instability
						situation_current_phase = situation_dynastic_cycle_phase_instability_conquest
					}
				}
			}
			trigger_if = {
				limit = { is_independent_ruler = no }
				liege = { has_realm_law = gallic_bureaucracy_2 }
			}
		}

		can_pass = {
			trigger_if = {
				limit = {
					NOT = { has_realm_law = gallic_bureaucracy_3 }
					is_independent_ruler = yes
				}
				has_realm_law = gallic_bureaucracy_1
				custom_description = {
					subject = root
					text = "has_gallic_bureaucracy_cooldown"
					NAND = {
						has_variable = gallic_bureaucracy_cooldown
						NOT = {
							culture = { has_innovation = innovation_all_things }
						}
					}
				}
			}
			trigger_if = {
				limit = {
					is_independent_ruler = no
					liege = { has_realm_law = gallic_bureaucracy_2 }
				}
				custom_tooltip = {
					text = gallic_bureaucracy_pending_law_change
					always = no
				}
			}
			can_pass_law_ca2_trigger = yes
		}

		pass_cost = {
			prestige = {
				if = {
					limit = {
						NOT = { has_realm_law = gallic_bureaucracy_3 }
					}
					add = increase_crown_authority_prestige_cost
				}
			}
		}
		
		on_pass = {
			# Increment cooldown breaker tally, if appropriate.
			calculate_authority_cooldown_break_effect = yes
			# Set regular cooldown.
			set_variable = {
				name = gallic_bureaucracy_cooldown
				years = @gallic_bureaucracy_cooldown_years_2
			}
			# Remove modifiers.
			remove_law_related_modifiers_effect = yes
		}

		ai_will_do = {
			if = {
				limit = { has_realm_law = gallic_bureaucracy_1 }
				value = 1
			}
		}
	}

			barons_and_minor_landholders_opinion = -20
			glory_hound_opinion = -25

			zealot_opinion = -20

			vassal_tax_contribution_mult = 0.15
			vassal_levy_contribution_mult = 0.15
			republic_government_tax_contribution_add = 0.05

		flag = uses_gallic_bureaucracy
		flag = max_authority_level

		flag = vassal_all_wars_banned_permanent
		flag = can_have_tributaries
		
		can_keep = {

			trigger_if = {

		can_pass = {

				custom_description = {
					subject = root

					NAND = {
						has_variable = gallic_bureaucracy_cooldown
						NOT = {
							culture = { has_innovation = innovation_all_things }
						}
					}
				}
			}
			trigger_if = {
				limit = {
					is_independent_ruler = no
					liege = { has_realm_law = gallic_bureaucracy_3 }
				}
				custom_tooltip = {
					text = gallic_bureaucracy_pending_law_change
					always = no
				}
			}
			can_pass_law_ca3_trigger = yes
		}
		
		pass_cost = {
			prestige = {
				if = {
					limit = {
						NOT = { has_realm_law = gallic_bureaucracy_3 }
					}
					add = increase_crown_authority_prestige_cost
				}
			}
		}
		
		on_pass = {
			# Increment cooldown breaker tally, if appropriate.
			calculate_authority_cooldown_break_effect = yes
			# Set regular cooldown.
			set_variable = {
				name = gallic_bureaucracy_cooldown
