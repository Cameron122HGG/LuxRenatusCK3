	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = language_south_germanic
		}
	}
	ai_will_do = {
		value = 10
		if = {
			limit = { has_cultural_pillar = language_south_germanic }
			multiply = 10
		}
	}
	
	color = { 0.45 0.3 0.3 }
}
