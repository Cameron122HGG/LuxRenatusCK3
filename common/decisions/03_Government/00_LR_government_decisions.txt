convert_high_tribal_to_feudalism_decision = {
	picture = {
		trigger = {
			#has_tgp_dlc_trigger = yes
			liege = {
				culture = { has_cultural_pillar = heritage_frankish }
			}
		}
		reference = "gfx/interface/illustrations/decisions/tgp_dominance.dds"
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	desc = convert_high_tribal_to_feudalism_decision_desc

	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	is_shown = {
		government_has_flag = government_is_high_tribal
		OR = {
			liege = this
			AND = {
				liege != this
				liege = {
					NOR = {
						government_has_flag = government_is_high_tribal
						government_has_flag = government_is_tribal
						government_has_flag = government_is_nomadic
					}
				}
			}
		}
	}

	is_valid = {
		has_realm_law = crown_authority_2
		custom_description = {
			text = faith_is_reformed
			subject = root
			faith = { NOT = { has_doctrine_parameter = unreformed } }
		}
		prestige_level >= 1
		custom_description = { #You need to have 70% of all (non-regional/unique) tribal innovations!
			text = has_all_tribal_innovations
			subject = root.culture
			culture = {
				calc_true_if = {
					amount >= 9
					has_innovation = innovation_motte
					has_innovation = innovation_catapult
					has_innovation = innovation_barracks
					has_innovation = innovation_mustering_grounds
					has_innovation = innovation_bannus
					has_innovation = innovation_quilted_armor
					has_innovation = innovation_gavelkind
					has_innovation = innovation_crop_rotation
					has_innovation = innovation_city_planning
					has_innovation = innovation_casus_belli
					has_innovation = innovation_plenary_assemblies
					has_innovation = innovation_ledger
					has_innovation = innovation_development_01
					has_innovation = innovation_currency_01
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = permafrost_feudal_ban
			NOT = { culture = { has_cultural_parameter = permafrost_cannot_become_feudal } }
		}
	}

	cost = {
		prestige = 1000
	}

	effect = {
		#gain_legitimizing_legend_seed_tooltip_effect = yes		
		if = {
			limit = {
				liege != this
				OR = {
					liege = { government_has_flag = government_is_classical }
					AND = {
						liege = { government_has_flag = government_is_classical }
						faith.religion = { is_in_family = rf_graeco_roman }
					}
				}
			}
			change_government = classical_government
		}
		else_if = {
			limit = {
				liege != this
				OR = {
					liege = { government_has_flag = government_is_despotic }
					AND = {
						liege = { government_has_flag = government_is_despotic }
						faith.religion = { is_in_family = rf_archaic }
					}
				}
			}
			change_government = despotic_government
		}
		else_if = {
			limit = {
				liege != this
				OR = {
					liege = { has_government = iqta_government }
					AND = {
						liege = { has_government = iqta_government }
						religion = religion:islam_religion
					}
				}
			}
			change_government = iqta_government
		}		
		##Independent
		else_if = {
			limit = {
				liege = this
				faith.religion = { is_in_family = rf_graeco_roman }
			}
			change_government = classical_government
		}
		else_if = {
			limit = {
				liege = this
				faith.religion = { is_in_family = rf_archaic }
			}
			change_government = despotic_government
		}
		else_if = {
			limit = {
				liege = this
				faith = { religion_tag = islam_religion }
			}
			change_government = iqta_government
		}		
		else = { change_government = western_feudal_government }
		
		##Convert Vassals
		custom_tooltip = {
			text = LR_high_tribal_vassal_adoption
			if = {
				limit = {
					government_has_flag = government_is_classical
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = classical_government
				}
			}
			else_if = {
				limit = {
					government_has_flag = government_is_despotic
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = despotic_government
				}
			}
			else_if = {
				limit = {
					has_government = iqta_government
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = iqta_government
				}
			}		
			else = {
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = western_feudal_government
				}
			}
		}
		
		save_scope_as = actor
		trigger_event = LR_gov_events.0001
		hidden_effect = {
			every_vassal = {
				trigger_event = {
					id = LR_gov_events.0002
					days = 3
				}
			}
		}
	}

	ai_potential = {
		NOT = { mpo_has_gok_mongol_empire_trigger = yes } #Stop the Mongols from slowing their conquest. 
	}

	ai_will_do = {
		base = 100
	}
}

convert_classical_decision = {
	picture = {
		trigger = {
			#has_tgp_dlc_trigger = yes
			liege = {
				culture = { has_cultural_pillar = heritage_latin }
			}
		}
		reference = "gfx/interface/illustrations/decisions/tgp_dominance.dds"
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	desc = convert_classical_decision_desc

	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	is_shown = {
		OR = {
			has_government = western_feudal_government 
			has_government = iqta_government
			AND = {
				is_independent_ruler = no
				has_government = despotic_government
				liege = {
					has_government = classical_government
				}
			}
		}
		NOT = { government_has_flag = government_is_high_tribal }
		OR = {
			AND = {
				liege = this
				OR = {
					faith = { has_doctrine = lux_doctrine_graeco_roman_ascendant }
					faith = { has_doctrine = graeco_roman_special_hostility_doctrine }
				}	
				NOR = {
					has_government = classical_government
					has_government = despotic_government
				}
			}
			AND = {
				NOT = { liege = this }
				NOT = { has_government = classical_government }
				liege = {
					has_government = classical_government
				}
			}
		}
	}

	is_valid = {
		OR = {
			faith = { has_doctrine = lux_doctrine_graeco_roman_ascendant }
			faith = { has_doctrine = graeco_roman_special_hostility_doctrine }
			trigger_if = {
				limit = {
					is_independent_ruler = no
				}
				liege = {
					has_government = classical_government			
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = permafrost_feudal_ban
			NOT = { culture = { has_cultural_parameter = permafrost_cannot_become_feudal } }
		}
	}

	cost = {
		prestige = 500
	}

	effect = {
		#gain_legitimizing_legend_seed_tooltip_effect = yes		
		change_government = classical_government
		custom_tooltip = {
			text = LR_high_tribal_vassal_adoption			
			every_vassal_or_below = {
				limit = { 
					OR = {
						has_government = western_feudal_government 
						has_government = iqta_government
					}
				}
				change_government = classical_government
			}
		}
		save_scope_as = actor
		trigger_event = LR_gov_events.0003
		hidden_effect = {
			every_vassal = {
				trigger_event = {
					id = LR_gov_events.0004
					days = 3
				}
			}
		}		
	}

	ai_potential = {
		NOT = { mpo_has_gok_mongol_empire_trigger = yes } #Stop the Mongols from slowing their conquest. 
	}

	ai_will_do = {
		base = 100
	}
}
