# Re-establish Theodosian Borders (western)  
retake_western_provinces_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	ai_check_interval = 120

	sort_order = 50

	desc = retake_western_provinces_decision_desc
	selection_tooltip = retake_western_provinces_decision_tooltip
	
	decision_group_type = roman

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		has_title = title:e_spqr
		
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:retake_western_provinces_decision
			}
		}
	}

	is_valid = {
		completely_controls_region = custom_honorian_wre
		custom_tooltip = {
			text = western_kingdoms_held_or_vassals_tt
			
			title:k_roman_italia_annonaria.holder = { liege = root }
			title:k_roman_italia_suburbicaria.holder = { liege = root }
			title:k_roman_illyricum.holder = { liege = root }
			title:k_roman_galliarum.holder = { liege = root }
			title:k_roman_viennensis.holder = { liege = root }
			title:k_roman_africa.holder = { liege = root }
			title:k_roman_mauretania.holder = { liege = root }
			title:k_roman_hispania.holder = { liege = root }
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		NOT = { exists = involved_activity }
		is_imprisoned = no
		has_contagious_deadly_disease_trigger = no
	}

	cost = {
		prestige = {
			value = 1000		
		}
	}

	effect = {
		show_as_tooltip = {
			retake_western_provinces_decision_effect = yes
		}
		trigger_event = {
			id = lux_major_decisions.0004
		}
		#Can only be done once
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:retake_western_provinces_decision
		}
	}

	ai_potential = {
		has_title = title:e_spqr
		is_at_war = no
		NOT = { exists = involved_activity }

restore_wrh_roman_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	desc = restore_wrh_roman_empire_decision_desc
	selection_tooltip = restore_wrh_roman_empire_decision_tooltip
	decision_group_type = roman

	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		is_roman_emperor_excluding_eastern_rome_trigger = yes
		NOT = { primary_title = title:e_hre }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_roman_empire
			}
		}
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_republic
			government_has_flag = government_is_clan
			government_allows = administrative
		}
	}

	is_valid = {
		prestige_level >= very_high_prestige_level
		
		custom_tooltip = { #Only available after WRE borders restored
            text = wre_borders_restored_tt
            is_target_in_global_variable_list = { 
            	name = unavailable_unique_decisions
            	target = flag:retake_western_provinces_decision
        	}
        } 	
		
		completely_controls = title:d_latium
		completely_controls = title:d_lombardia
		completely_controls = title:d_sicily
		completely_controls = title:d_valois
		completely_controls = title:d_aquitaine
		completely_controls = title:d_brittany
		completely_controls = title:d_toledo
		completely_controls = title:d_tunis
		completely_controls = title:d_slavonia
		completely_controls = title:d_turnovo
		completely_controls = title:d_thrace
		completely_controls = title:d_epirus
		completely_controls = title:d_athens
		
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes

		OR = {
			religion = { is_in_family = rf_abrahamic }
			religion = religion:hellenism_religion
			culture = { has_cultural_pillar = heritage_latin }
			culture = { has_cultural_pillar = heritage_byzantine }
		}
	}

	effect = {
		custom_tooltip = restore_wrh_roman_empire_decision_effect_tooltip
		show_as_tooltip = {
			create_wrh_roman_empire_scripted_effect = yes
		}
		if = {
			limit = {
				has_ep3_dlc_trigger = yes
			}
			trigger_event = lux_major_decisions.0015
		}
		else = {
			trigger_event = roman_restoration.0001
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_roman_empire
			}
			set_global_variable = {
				name = flag_restored_roman_empire
				value = root
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# Roman Renaissance #DONE 
roman_renaissance_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/ep3_decision_hold_triumph.dds"
    }
    ai_check_interval = 120

    sort_order = 50

    desc = roman_renaissance_decision_desc
    selection_tooltip = roman_renaissance_decision_tooltip
    
    decision_group_type = roman

    is_shown = {
        #has_ep3_dlc_trigger = yes
        is_ruler = yes
        is_playable_character = yes
        is_roman_emperor_excluding_byzantium_trigger = yes
        NOT = {
            this = culture:roman
        }
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:roman_renaissance_decision
            }
        }
    }
    is_valid = {
    	completely_controls_region = ghw_region_italy
        custom_description = {
            text = capital_in_italy_region_tooltip
            capital_province = {
                geographical_region = ghw_region_italy
            }
        }

        #Should have high prestige level.
        prestige_level >= 4

        #Should be a romman descendent
        OR = {
			religion = { is_in_family = rf_abrahamic }
			religion = religion:hellenism_religion
			culture = { has_cultural_pillar = heritage_latin }
			culture = { has_cultural_pillar = heritage_byzantine }
		}

        trigger_if = {
            limit = {
                OR = {
                    any_realm_county = {
                        culture = culture:roman
                    }
                    culture.culture_head = {
                        any_realm_county = {
                            culture = culture:roman
                        }
                    }
                    culture:roman = {
                        exists = culture_head
                    }
                }
            }
            culture = {
                cultural_acceptance = {
                    target = culture:roman
                    value >= 40
                }
            }
        }
    }   

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:roman_renaissance_decision
        }
        show_as_tooltip = {
            embrace_roman_culture_effect = yes
        }

        trigger_event = lux_major_decisions.0005
    }

    is_valid_showing_failures_only = {
        is_at_war = no
        NOT = { exists = involved_activity }
        is_imprisoned = no
        has_contagious_deadly_disease_trigger = no
    }

    ai_potential = {
        OR = {
            has_title = title:e_spqr
            has_title = title:h_roman_empire
        }
        
        is_at_war = no
        NOT = { exists = involved_activity }
        is_imprisoned = no
        has_contagious_deadly_disease_trigger = no
    }

    ai_will_do = {
        base = 100
    }

    cost = {
        prestige = {
            value = 1000        
        }
    }
}
