LR_diadochi_seleucid_kingdom_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
    }

    desc = LR_diadochi_seleucid_kingdom_desc
    selection_tooltip = LR_diadochi_seleucid_kingdom_tooltip
	confirm_text = LR_diadochi_seleucid_kingdom_confirm
    decision_group_type = major

    ai_check_interval = 120

    is_shown = {
        is_ruler = yes
        is_playable_character = yes
		NOT = { exists = title:e_seleucid_kingdom.holder }
		culture = culture:hellenistic

		NOT = { has_title = title:e_antigonid_empire }
		OR = {
		NOT = { has_primary_title = title:e_byzantium }
		NOT = { has_primary_title = title:e_antipatrid_empire }
		NOT = { has_primary_title = title:e_ptolemaic_kingdom }
		}

		NOT = { dynasty = dynasty:dynasty_hellenic_antigonid }
		NOT = { dynasty = dynasty:dynasty_hellenic_antipater }
		NOT = { dynasty = dynasty:dynasty_hellenic_ptolemaios}

		OR = {
		dynasty = dynasty:dynasty_hellenic_seleucid
		dynasty = dynasty:dynasty_hellenic_seleukos
		}

        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_lux_seleucid_reborn
            }
        }
    }

    is_valid = {
        is_landed = yes
		#is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
		NOT = { faith = { religion_tag = islam_religion } }

		culture = culture:hellenistic

		has_realm_law = crown_authority_3

		OR = {
		   NOT = { has_title = title:e_byzantium }
		   NOT = { exists = title:e_byzantium.holder }
		}

		OR = {
		dynasty = dynasty:dynasty_hellenic_seleucid
		dynasty = dynasty:dynasty_hellenic_seleukos
		}

		OR = {
			faith = faith:messalian
			faith = faith:nestorian
			faith = { religion_tag = solar_religion }
			faith = { religion_tag = zoroastrianism_religion }
			faith = { religion_tag = hellenism_religion }
		}

		prestige_level >= 3


		completely_controls = title:k_persia
		completely_controls = title:k_syria
		completely_controls = title:k_osroene
		completely_controls = title:k_mesopotamia

		OR = {
		    completely_controls = title:k_armenia
			completely_controls = title:k_jerusalem
		}
    }

    cost = {
         gold = 6000
    }

    effect = {
		custom_tooltip = LR_diadochi_seleucid_kingdom_effect_tooltip

	hidden_effect = {
	    change_government = administrative_government
	}

    hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_lux_seleucid_reborn
            }
            set_global_variable = {
                name = flag_lux_seleucid_reborn
                value = root
            }
        }

		trigger_event = LR_diadochi_kingdoms_events.1100
    }

    is_available = {
        always = yes
    }

    ai_will_do = {
        base = 1000000
    }
}
