                }
				OR = {
					title:k_genoa.holder = root
					title:k_genoa.holder = { liege = root }
				}
            }
        }

    }			

    effect = {
    		if = {
			limit = {
			has_dlc_feature = all_under_heaven
			}
			change_government = roman_autocratic_government
			save_scope_as = actor
			#restore_gallic_empire_provinces_gallia_viennensis_decision_effect = yes
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		every_vassal_or_below = {
			if = {
				limit = {
					has_dlc_feature = all_under_heaven
					OR = {
						government_has_flag = government_is_classical_administrative
						AND = {
							government_has_flag = government_is_roman
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_autocratic_government
			}
			else_if = {
				limit = {
					NOT = { has_dlc_feature = all_under_heaven }
					OR = {
						government_has_flag = government_is_classical_feudal
						AND = {
							government_has_flag = government_is_roman
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_autocratic_government
			}

        custom_tooltip = restore_western_diocese_system_unlock_tooltip

			name = num_of_provinces_restored_wre

		}

            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:western_roman_diocese_restored
            }

        prestige = {

        }

        is_at_war = no
        NOT = { exists = involved_activity }
        is_imprisoned = no
        has_contagious_deadly_disease_trigger = no

        OR = {
            has_title = title:e_spqr

        }
