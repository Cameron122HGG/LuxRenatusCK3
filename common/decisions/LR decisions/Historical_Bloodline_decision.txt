historical_bloodlines_decision = { 
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = bloodline

	ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		NOT = {	
			is_target_in_global_variable_list = {	
				name = unavailable_bloodline_decisions
				target = flag:flag_bloodline
			}
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "SET_BLOODLINES"
		show_from_start = yes

		item = { # CK2
			value = CK2_bloodlines
			is_valid = {
			}
			localization = CK2_bloodlines
			ai_chance = { value = 0 }
		}
		item = { # CK3
			value = CK3_bloodlines
			is_valid = {
			}
			localization = CK3_bloodlines
			ai_chance = { value = 0 }
		}

		item = { # 个人
			value = historical_character_bloodlines
			is_shown = {
			}
			localization = historical_character_bloodlines
			ai_chance = { value = 0 }
		}

		item = { # 事件角色
			value = historical_event_character_bloodlines
			is_shown = {
				is_historical_character = yes
			}
			localization = historical_event_character_bloodlines
			ai_chance = { value = 0 }
		}

		item = { # 宗族
			value = historical_family_bloodlines
			is_shown = {
			}
			localization = historical_family_bloodlines
			ai_chance = { value = 0 }
		}

		item = { # 非历史
			value = bloodlines_unhistorical
			is_shown = {
			}
			localization = bloodlines_unhistorical
			ai_chance = { value = 0 }
		}
	}

	effect = {
		if = {
			limit = {
				scope:CK2_bloodlines = yes
			}
			set_global_variable = {
				name = create_CK2_bloodline
				value = 1
			}
			trigger_event = historical_bloodline.1
			custom_tooltip = {
				text = CK2_bloodlines_tooltip
			}
		}
		else_if = {
			limit = {
				scope:CK3_bloodlines = yes
			}
			set_global_variable = {
				name = create_CK3_bloodline
				value = 1
			}
			trigger_event = historical_bloodline.5
			custom_tooltip = {
				text = CK3_bloodlines_tooltip
			}
		}
		else_if = {
			limit = {
				scope:historical_character_bloodlines = yes
			}
			set_global_variable = {
				name = create_historical_character_bloodline
				value = 1
			}
			trigger_event = historical_bloodline.2
			custom_tooltip = {
				text = historical_character_bloodlines_tooltip
			}
		}
		else_if = {
			limit = {
				scope:historical_family_bloodlines = yes
			}
			set_global_variable = {
				name = create_historical_family_bloodline
				value = 1
			}
			trigger_event = historical_bloodline.3
			custom_tooltip = {
				text = historical_family_bloodlines_tooltip
			}
		}
		else_if = {
			limit = {
				scope:bloodlines_unhistorical = yes
			}
			set_global_variable = {
				name = create_unhistorical_character_bloodline
				value = 1
			}
			trigger_event = historical_bloodline.4
			custom_tooltip = {
				text = bloodlines_unhistorical_tooltip
			}
		}
		else_if = {
			limit = {
				scope:historical_event_character_bloodlines = yes
			}
			trigger_event = historical_bloodline.5
			custom_tooltip = {
				text = historical_event_character_bloodlines_tooltip
			}
		}
	}
}
