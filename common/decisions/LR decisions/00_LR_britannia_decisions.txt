
###Unite England - Anglo-Saxon
unify_england_anglo_saxon_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	desc = unify_england_anglo_saxon_decision_desc

	is_shown = {
		OR = {
			culture = { has_cultural_pillar = heritage_west_germanic }
			culture = { has_cultural_pillar = heritage_north_germanic }
			culture = { has_cultural_pillar = heritage_central_germanic }
		}
		primary_title.tier < tier_empire
		primary_title.title_capital_county.title_province = { geographical_region = ghw_region_england }
		NOT = { has_global_variable = unified_england_variable }
	}

	is_valid = {
		is_independent_ruler = yes
		custom_tooltip = {
			text = primary_title_is_kingdom_tooltip
			primary_title.tier = tier_kingdom
		}
		completely_controls_region = ghw_region_england
		prestige_level >= 3
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	cost = { prestige = 500 }

	effect = {
		show_as_tooltip = {
			get_title = title:k_united_england
		}
		hidden_effect = {
			root = { save_scope_as = united_england_founder }
			primary_title.de_jure_liege = { save_scope_as = former_imperial_liege }
			root.primary_title = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_duchy }
					set_de_jure_liege_title = title:k_united_england
				}
			}
			title:k_united_england = { set_de_jure_liege_title = scope:former_imperial_liege }
		
			create_title_and_vassal_change = {
				type = created
				save_scope_as = title_change
			}
			title:k_united_england = {
				change_title_holder = {
					holder = scope:united_england_founder
					change = scope:title_change
				}
			}
			resolve_title_and_vassal_change = scope:title_change
			every_duchy = {
				limit = { 
					title_capital_county.title_province = { geographical_region = ghw_region_england } 
				}
				set_de_jure_liege_title = title:k_united_england
			}
			every_kingdom = {
				limit = { 
					title_capital_county.title_province = { geographical_region = ghw_region_england } 
					#this = exists
					NOT = { this = title:k_united_england } 
				}
				holder = { 
					destroy_title = prev 
				}
			}
			set_global_variable = {
				name = unified_england_variable
				value = yes
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
