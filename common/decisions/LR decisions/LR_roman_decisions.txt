
# Restore the Western Roman Empire if its gone.
LR_restore_western_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	desc = LR_restore_western_empire_decision_desc
	selection_tooltip = LR_restore_western_empire_decision_tooltip
	decision_group_type = roman


	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	is_shown = {
		culture = { has_cultural_pillar = heritage_latin }
		is_ruler = yes
		is_playable_character = yes
		highest_held_title_tier > 2 #Dukes and above
		capital_province = { geographical_region = ghw_region_italy }
		NOT = { exists = title:e_spqr.holder }
		NOT = { exists = title:h_western_roman_empire.holder }
		NOT = { exists = title:h_roman_empire.holder }

		#Just to guarantee
		#is_roman_emperor_trigger = no
	}

	is_valid = {
		#NOT = { exists = title:e_hre }
		NOT = { exists = title:e_spqr }
		NOT = { exists = title:h_western_roman_empire }
		NOT = { exists = title:h_roman_empire }
		prestige_level >= very_high_prestige_level
		has_title = title:k_rome
		completely_controls_region = custom_spqr_imperial_core
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = LR_restore_western_empire_decision_effect_tooltip
		show_as_tooltip = {
			restore_western_rome_scripted_effect = yes
		}

		if = {
			limit = { has_ep3_dlc_trigger = yes }			
			convert_to_administrative_from_feudalism_effect = { 
				GOVERNMENT_TO_ADOPT = flag:administrative 
			}
		}

		trigger_event = {
			id = LR_major_decisions.0003
		}

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_western_roman_empire
			}
			set_global_variable = {
				name = flag_restored_western_roman_empire
				value = root
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

		}

	}
	decision_group_type = roman
	sort_order = 50

	is_shown = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:merged_pontus
			}
		}
		OR = {
        culture = { has_cultural_pillar = heritage_byzantine }
        culture = { has_cultural_pillar = heritage_hellenistic_greek }
        has_title = title:e_pontus
		has_title = title:e_byzantium
		}
		NOR = {
			title:e_pontus = { is_titular = yes }
			title:e_byzantium = { is_titular = yes }
		}
	}

	is_valid = {
		has_title = title:e_pontus
		has_title = title:e_byzantium
		top_liege = this
		prestige_level >= 3
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:merged_pontus
		}

		save_scope_as = rightful_liege

		add_legitimacy_effect = { LEGITIMACY = massive_legitimacy_gain }
		dynasty ?= { add_dynasty_prestige = massive_dynasty_prestige_value }

			limit = {

			}
		}

				SECONDARY = e_pontus

			}
		}

		every_vassal_or_below = {
			trigger_event = major_decisions.3200
		}
	}

	cost = {
		prestige = {
			if = {
				limit = {
					is_ai = no
				}
				value = 1000
			}
			else = {

			}
		}
	}

restore_gallic_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_restore_gallic_empire.dds"
	}
	decision_group_type = roman

	desc = restore_gallic_empire_decision_desc
	selection_tooltip = restore_gallic_empire_decision_tooltip

	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		is_landed_or_landless_administrative = yes
		OR = {
			government_has_flag = government_is_gaul_feudal
			government_has_flag = government_is_gaul_administrative
			#government_has_flag = government_is_roman_imperial
			#government_has_flag = government_is_administrative
		}
		culture = { has_cultural_pillar = heritage_latin }
		any_realm_county = {
			title_province = {geographical_region = LR_world_europe_gaul}
		  }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:restore_gallic_empire_decision
			}
		}
		highest_held_title_tier <= tier_empire
	}

	is_valid = {
		highest_held_title_tier = tier_empire
		prestige_level >= high_prestige_level
		OR = {
			is_independent_ruler = yes
			AND = {
				highest_held_title_tier = tier_empire
				liege = {
					OR = {
						government_has_flag = government_is_classical_administrative
						government_has_flag = government_is_roman_imperial
						government_has_flag = government_is_gaul_administrative
					}
				}
			}
		}
		completely_controls_region = LR_world_europe_gaul
		#completely_controls_region = custom_roman_britannia

	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	effect = {
		if = {
			limit = {
			has_dlc_feature = all_under_heaven
			}
			change_government = roman_imperial_government
			save_scope_as = actor
			restore_gallic_empire_provinces_gallia_viennensis_decision_effect = yes
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		every_vassal_or_below = {
			if = {
				limit = {
					has_dlc_feature = all_under_heaven
					OR = {
						government_has_flag = government_is_gaul_administrative
						AND = {
							government_has_flag = government_is_roman
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_imperial_government
			}
			else_if = {
				limit = {
					NOT = { has_dlc_feature = all_under_heaven }
					OR = {
						government_has_flag = government_is_gaul_feudal
						AND = {
							government_has_flag = government_is_roman
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_imperial_government
			}
		}
		title:e_gallic_empire = {
			save_scope_as = title_e_gallic_empire
		}
		title:e_gallic_empire = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
			save_scope_as = title_e_gallic_empire
		}
		resolve_title_and_vassal_change = scope:title_change
		if = {
			limit = { exists = title:k_sacral_king_gaul_throne.holder }
			destroy_title = title:k_sacral_king_gaul_throne
		}
		if = {
			limit = { exists = title:e_high_kingdom_gaul.holder }
			destroy_title = title:e_high_kingdom_gaul
		}
		
		#trigger_event = {
		#	id = province_events.0040
		#	days = 2
		#}
		add_prestige = 2000
		
		hidden_effect = {

			#restore_roman_provinces_gallia_viennensis_decision_effect = yes

			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:restore_gallic_empire_decision
			}
		}
	}

	cost = {
		gold = 1500
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
