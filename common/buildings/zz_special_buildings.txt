roman_plaza_01 = {
	construction_time = slow_construction_time

	can_construct_potential = {
		county.holder.culture = { has_cultural_parameter = roman_plaza }
		building_requirement_tribal = no
	}
	
	can_construct_showing_failures_only = {
		county.holder.culture = { has_cultural_parameter = roman_plaza }
	}
	
	is_enabled = {
		county.holder = { has_title = prev.duchy }
	}
	show_disabled = yes
	
	cost_gold = expensive_building_tier_3_cost
	
	character_modifier = {
		men_at_arms_recruitment_cost = -0.05
		monthly_prestige_gain_mult = 0.05
		legitimacy_gain_mult = 0.05
		monthly_legitimacy_add = 5
		monthly_influence = 1
	}
	
	next_building = roman_plaza_02

	type_icon = "icon_building_legendary_statue.dds"
	
	type = duchy_capital
	
	ai_value = {
		base = 20
		modifier = {
			factor = 2
			scope:holder.capital_province = this
		}
		modifier = { # Fill all building slots before going for duchy buildings
			factor = 0
			free_building_slots > 0
		}
		modifier = { # Less likely when the AI should build regular buildings first
			factor = 0.1
			scope:holder = { ai_should_focus_on_building_in_their_capital = yes }
		}
	}
}
