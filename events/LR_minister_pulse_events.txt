###############################################################
# Different Administrative — minister side-effect pulse events
#
# Triggered quarterly (~25% chance per minister, so ≈ once / year
# per filled ministry slot) from the admin top liege's
# quarterly_playable_pulse. Each event picks an outcome from an
# internal random_list, applies effects to scope:minister and
# root (top liege), and shows a single-option letter to the liege.
#
# Scope contract:
#   root        = top liege (council owner)
#   scope:minister = the councillor character in that ministry
###############################################################

namespace = LR_minister_pulse

###############################################################
# 1 — Minister of Personnel (diplomacy)
###############################################################
LR_minister_pulse.1 = {
	type = letter_event
	opening = LR_minister_pulse.1.opening
	sender = scope:minister
	title = LR_minister_pulse.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = LR_pulse_outcome_talent }
				desc = LR_minister_pulse.1.desc.talent
			}
			triggered_desc = {
				trigger = { has_variable = LR_pulse_outcome_scandal }
				desc = LR_minister_pulse.1.desc.scandal
			}
			triggered_desc = {
				trigger = { has_variable = LR_pulse_outcome_smooth }
				desc = LR_minister_pulse.1.desc.smooth
			}
			triggered_desc = {
				trigger = { has_variable = LR_pulse_outcome_coup }
				desc = LR_minister_pulse.1.desc.coup
			}
			desc = LR_minister_pulse.1.desc.smooth
		}
	}
	theme = court
	left_portrait = {
		character = scope:minister
		animation = personality_bold
	}

	immediate = {
		random_list = {
			# Talent discovered — scaled by minister diplomacy
			40 = {
				modifier = { add = scope:minister.diplomacy factor = 0.05 }
				scope:minister = { set_variable = { name = LR_pulse_outcome_talent days = 30 } }
				add_prestige = 75
			}
			# Smooth diplomacy
			25 = {
				scope:minister = {
					set_variable = { name = LR_pulse_outcome_smooth days = 30 }
					add_prestige = 50
				}
			}
			# Major coup
			10 = {
				modifier = { add = scope:minister.diplomacy factor = 0.03 }
				scope:minister = {
					set_variable = { name = LR_pulse_outcome_coup days = 30 }
					add_prestige = 100
				}
				add_prestige = 100
				add_legitimacy = 25
			}
		}
	}

	option = {
		name = LR_minister_pulse.option.acknowledge
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_talent } }
	}
	option = {
		name = LR_minister_pulse.option.acknowledge_negative
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_scandal } }
		add_opinion = {
			modifier = displeased_opinion
			target = scope:minister
			opinion = -10
		}
	}
	option = {
		name = LR_minister_pulse.option.well_done
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_smooth } }
		add_character_modifier = {
			modifier = LR_minister_smooth_diplomacy_modifier
			years = 5
		}
	}
	option = {
		name = LR_minister_pulse.option.outstanding
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_coup } }
		add_opinion = {
			modifier = pleased_opinion
			target = scope:minister
			opinion = 15
		}
	}

	after = {
		scope:minister = {
			remove_variable = LR_pulse_outcome_talent
			remove_variable = LR_pulse_outcome_scandal
			remove_variable = LR_pulse_outcome_smooth
			remove_variable = LR_pulse_outcome_coup
		}
	}
}

			}
			triggered_desc = {
				trigger = { has_variable = LR_pulse_outcome_reform }

			}

				trigger = { has_variable = LR_pulse_outcome_corrupt }

				trigger = { has_variable = LR_pulse_outcome_blunder }

		}
	}
	theme = court
	left_portrait = {
		character = scope:minister
		animation = personality_rational
	}

	immediate = {
		random_list = {
			40 = {

				modifier = { add = scope:minister.learning factor = 0.04 }

				add_tyranny = -10

				modifier = { add = scope:minister.learning factor = 0.04 }
				scope:minister = { set_variable = { name = LR_pulse_outcome_reform days = 30 } }
				scope:minister = { add_prestige = 50 }

		}

	}

	option = {
		name = LR_minister_pulse.option.well_done
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_tyranny_reduced } }
	}
	option = {
		name = LR_minister_pulse.option.outstanding
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_reform } }
		add_character_modifier = {
			modifier = LR_minister_legal_reformer_modifier
			years = 10
		}
		add_opinion = {
			modifier = pleased_opinion
			target = scope:minister
			opinion = 10
		}
	}
	option = {
		name = LR_minister_pulse.option.acknowledge_negative
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_corrupt } }
	}
	option = {
		name = LR_minister_pulse.option.disappointed
		trigger = { scope:minister = { has_variable = LR_pulse_outcome_blunder } }
		add_character_modifier = {
			modifier = LR_minister_judicial_blunder_modifier
			years = 2
		}
		add_opinion = {
			modifier = displeased_opinion
			target = scope:minister
			opinion = -15
		}
	}

	after = {
		scope:minister = {
			remove_variable = LR_pulse_outcome_tyranny_reduced
			remove_variable = LR_pulse_outcome_reform
			remove_variable = LR_pulse_outcome_corrupt
			remove_variable = LR_pulse_outcome_blunder
		}
