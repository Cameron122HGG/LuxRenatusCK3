fortify_aegean.0008 = {
	type = character_event
	title = fortify_aegean.0008.t
	desc = fortify_aegean.0008.desc
	override_background = { reference = throne_room }
	theme = emperor
	left_portrait = { character = root }
	option = {
		name = fortify_aegean.0008.a
		set_global_variable = { name = cretan_raids months = 1200 }
		title:c_rhodos = { add_county_modifier = { modifier = cretan_raid_devastation_modifier months = 600} }
		title:c_naxos = { add_county_modifier = { modifier = cretan_raid_devastation_modifier months = 600} }
		title:c_chios = { add_county_modifier = { modifier = cretan_raid_devastation_modifier months = 600} }
		trigger_event = {
			id = fortify_aegean.0009
			years = 4
		}
		title:d_krete.holder = {
			trigger_event = {
				id = fortify_aegean.0025
			}
		}
	}
}

fortify_aegean.0009 = {
	type = character_event
	hidden = yes
	title = fortify_aegean.0009.t
	desc = fortify_aegean.0009.desc
	override_background = { reference = throne_room }
	theme = emperor
	left_portrait = { character = root }
	option = {
		name = fortify_aegean.0009.a
		if = {
			limit = {
				has_global_variable = cretan_raids
				NOT = { title:d_krete = { holder.top_liege = root.top_liege } }
				NOT = {		
					title:d_krete = {
						religion = root.religion
					}
				}	
			}
			trigger_event = {
				id = fortify_aegean.0010
			}
		}
	}
}
