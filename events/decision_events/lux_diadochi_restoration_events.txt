LR_diadochi_kingdoms_events.1000 = {  #Successor Kingdoms
	type = character_event
	title = LR_diadochi_kingdoms_events.1000.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_global_variable = lux_flag_ptolemaic_kingdom
				}
				desc = LR_diadochi_ptolemaic_kingdom_desc
			}
			triggered_desc = {
				trigger = {
					has_global_variable = lux_flag_antigonid_kingdom
				}
				desc = LR_diadochi_antigonid_kingdom_desc
			}
			
			triggered_desc = {
				trigger = {
					has_global_variable = lux_flag_antipatrid_kingdom
				}
				desc = LR_diadochi_antipatrid_kingdom_desc
			}
		}
    }

	
	theme = crown
	
	left_portrait = {
		character = root
		animation = happiness
	}

	immediate = {
		#Same-faith non-empire tier rulers are over-awed.
		if = {
			limit = { faith = scope:scoped_emperor.primary_title.state_faith }
			play_music_cue = "mx_cue_epic_sacral_moment"
		}
		#Otherwise, this is a tacit statement of imperial threat.
		else = { play_music_cue = "mx_cue_combat_2" }
	}

	

	
	option = {
		name = LR_diadochi_kingdoms_events.1000.a
		trigger = { has_global_variable = lux_flag_ptolemaic_kingdom }
		add_prestige = 1250
		add_piety = 650
		
		dynasty = {
				add_dynasty_prestige = 800
		}
    }
    
	
	
	option = {
		name = LR_diadochi_kingdoms_events.1000.b
